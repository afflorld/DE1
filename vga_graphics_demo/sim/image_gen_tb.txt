library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;

entity tb_image_gen is
end tb_image_gen;

architecture sim of tb_image_gen is

    
    signal clk      : std_logic := '0';
    signal video_on : std_logic := '0';
    signal x_pos    : integer range 0 to 1023 := 0;
    signal y_pos    : integer range 0 to 511 := 0;
    signal btn_u    : std_logic := '0';
    signal btn_d    : std_logic := '0';
    signal btn_r    : std_logic := '0';
    signal rgb      : std_logic_vector(11 downto 0);

    
    constant CLK_PERIOD : time := 40 ns;

begin

    
    uut: entity work.image_gen
        port map (
            clk      => clk,
            video_on => video_on,
            x_pos    => x_pos,
            y_pos    => y_pos,
            btn_u    => btn_u,
            btn_d    => btn_d,
            btn_r    => btn_r,
            rgb      => rgb
        );

    
    clk_process : process
    begin
        clk <= '0';
        wait for CLK_PERIOD/2;
        clk <= '1';
        wait for CLK_PERIOD/2;
    end process;

    
    stim_proc: process
    begin
        
        video_on <= '0';
        wait for 100 ns;
        
                video_on <= '1';
        x_pos <= 300; 
        y_pos <= 220; 
        wait for 100 ns;

        
        x_pos <= 100;
        wait for 100 ns;

        
        btn_u <= '1';
        wait for 100 ns;
        x_pos <= 50; y_pos <= 50;
        wait for 100 ns;
        btn_u <= '0';

       
        btn_d <= '1';
        x_pos <= 0; y_pos <= 0; 
        wait for CLK_PERIOD;
        x_pos <= 100;
        wait for 100 ns;
        btn_d <= '0';

       
        btn_r <= '1';
        wait for CLK_PERIOD * 2;
        btn_r <= '0';

        
        wait;
    end process;

end sim;
